% Force both first and last name AND section headings to render in moderncv

% LaTeX's automatic hyphenation can break "chatbot" as "chat-bot" at a
% line-wrap boundary (a genuine inserted hyphen, invisible until you check
% the actual PDF text layer via pdftotext) - an ATS text-layer keyword
% search for "chatbot" could miss that instance. Exclude it globally.
\hyphenation{chatbot}

% Remove the top-level itemize bullet glyph (used for Core Competencies,
% Education & Certifications, Languages, References) - Raymond found the
% default (large, blue) redundant next to bold labels/company names. Do
% NOT use \usepackage{enumitem} + \begin{itemize}[label={}] for this -
% enumitem resets list spacing globally and silently adds a page. This
% targeted redefinition only removes the glyph.
\renewcommand*{\labelitemi}{}

% Single-line job header: Title | Company - Location, dates flush right.
% Replaces moderncv's default 2-line \cventry layout (Company+Location on
% one line, Title+Dates on the next) with Raymond's preferred single-line
% format to save vertical space. #1 title, #2 company, #3 location, #4
% dates (pass #4 empty for undated Earlier Career entries).
\newcommand{\cvoneline}[4]{%
  \noindent\textbf{#1} \normalfont | #2 -- #3%
  \ifthenelse{\equal{#4}{}}{}{\hfill\textit{#4}}%
  \par\vspace{1pt}%
}
% Job-accomplishment bullets under a \cvoneline entry need the dash marker
% moderncv normally reserves for level-2 lists (\labelitemii) - but since
% \cvoneline entries aren't nested inside an outer wrapper itemize (unlike
% moderncv's default \cventry pattern), these bullets are level-1 lists
% and would otherwise inherit the empty \labelitemi above. Wrap each such
% itemize block in {\dashitemize \begin{itemize}...\end{itemize}} - the
% surrounding group keeps the override from leaking into Core
% Competencies/Education/Languages/References, which stay markerless.
\newcommand{\dashitemize}{\renewcommand*{\labelitemi}{\strut\textcolor{color1}{\large\bfseries-}}}
% Markerless lists (Core Competencies, Education & Certifications,
% Languages, References) still inherit itemize's normal left-margin/
% label-width geometry even with an empty \labelitemi glyph - the text
% sits indented into blank space that used to hold the marker. Zero the
% list-geometry lengths locally (same \leftmargini-override approach as
% \dashitemize, not enumitem) so the text starts flush with the section
% heading - this also reduces word-wrap since the text column is wider.
\newcommand{\flushitemize}{\setlength{\leftmargini}{0pt}\setlength{\labelsep}{0pt}\setlength{\labelwidth}{0pt}\setlength{\itemindent}{0pt}\setlength{\listparindent}{0pt}}

% One-line "so what" synthesis under a role's bullets (Raymond's preferred
% "insight line" pattern, borrowed from an external template he liked).
% Use sparingly - most relevant/senior roles only, it costs vertical space.
% Icon is yellow (lightbulbyellow); text stays color1 to match section
% headers. Raymond rejected a dark amber - wanted true bright yellow.
\newcommand{\insight}[1]{\vspace{2pt}\par{\small\itshape\color{lightbulbyellow}\faIcon{lightbulb}\color{color1}\ #1}}
